USE msdb;

WITH SqlAgentJobs AS
(
    SELECT
        j.name AS JobName,
        LOWER(
            REPLACE(
                REPLACE(
                    CASE
                        WHEN LOWER(LEFT(j.name, 3)) = 'pl_' THEN SUBSTRING(j.name, 4, LEN(j.name))
                        ELSE j.name
                    END,
                    '_', ''
                ),
                '-', ''
            )
        ) AS JobMatchKey
    FROM dbo.sysjobs j
),
AdfPipelines AS
(
    SELECT DISTINCT
        ObjectName AS PipelineName,
        LOWER(
            REPLACE(
                REPLACE(
                    CASE
                        WHEN LOWER(LEFT(ObjectName, 3)) = 'pl_' THEN SUBSTRING(ObjectName, 4, LEN(ObjectName))
                        ELSE ObjectName
                    END,
                    '_', ''
                ),
                '-', ''
            )
        ) AS PipelineMatchKey
    FROM YourDatabase.dbo.AdfPipelineInfoResult
    WHERE ObjectType = 'Pipeline'
)
SELECT
    j.JobName,
    p.PipelineName,
    CASE
        WHEN p.PipelineName IS NULL THEN 'Not Found In ADF'
        ELSE 'Matched'
    END AS MatchStatus
FROM SqlAgentJobs j
LEFT JOIN AdfPipelines p
    ON j.JobMatchKey = p.PipelineMatchKey
ORDER BY
    MatchStatus,
    j.JobName;
