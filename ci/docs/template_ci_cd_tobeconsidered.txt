build_dotnet_core_template.yml AS BELOW

 

.build:

  stage: build

  tags: ["${RUNNER_TAG}"]

  image:

    name: ${BAR_REGISTRY}/mtb-docker/${COMPILER_VERSION}

    entrypoint: [""]

  variables:

    JOB_TOOL: dotnet

    JOB_FUNCTION: build

    JOB_IS_REQUIRED: "true"

    PROJECT_PLATFORM: "not_relevant"

  script:

    - !reference [.pipeline_scripts_linux, script]

    - !reference [.veracode_zip_list, script]

    - cp ${NUGET_AUTH} ${NUGET_CONFIG_PATH}

    - mkdir -p ${OUTPUT_PATH}

    - dotnet publish ${SOLUTION_PATH} ${BUILD_OPTIONS} -o ${OUTPUT_PATH}

    # Build output

    - cd ${OUTPUT_PATH} && zip -r ${LOCAL_ARTIFACT_NAME} .

    - mv "${CI_PROJECT_DIR}/${OUTPUT_PATH}/${LOCAL_ARTIFACT_NAME}" "${CI_PROJECT_DIR}/"

    # Script to set which files get packaged for Veracode.  If the VERACODE_FILE_LIST file type

    # variable is not set in the project repo, then it will default to '.' to load all files

    - log info "Following files added to Veracode upload - ${ITEM_LIST}"

    - zip -q -r ${LOCAL_VERACODE_ARTIFACT_NAME} ${ITEM_LIST}

    - mv "${CI_PROJECT_DIR}/${OUTPUT_PATH}/${LOCAL_VERACODE_ARTIFACT_NAME}" "${CI_PROJECT_DIR}/"

 

  artifacts:

    expire_in: 4 hour

    paths:

      # - veracode_upload-${PKG}-${CI_COMMIT_SHORT_SHA}.zip

      - ${LOCAL_ARTIFACT_NAME}

      - ${LOCAL_VERACODE_ARTIFACT_NAME}

 

  retry: !reference [.retry_rules]

 

.build_package:

  extends: .build

  rules:

    - !reference [.all_environment_rules, rules]

# Generate a db migration

.generate_migrate_db_template:

  image: ${BAR_REGISTRY}/mtb-docker-ga/${GENERATE_MIGRATE_DB_IMAGE}

  stage: build

  variables:

    JOB_TOOL: db_migration

    JOB_FUNCTION: ef_migrations

    PROJECT_PLATFORM: "not_relevant"

  tags:

    - 'cicd-k8s'

  script:

    - !reference [.pipeline_scripts_linux, script]

    - cp ${NUGET_AUTH} ${NUGET_CONFIG_PATH}

    - dotnet ef migrations script --project ${PROJECT_FILE} --output ${MIGRATION_FILE_NAME} --idempotent ${VERBOSITY_LEVEL} --context ${CONTEXT}

    - curl -u ${BAR_USER}:${BAR_TOKEN} -X PUT "${SQL_ARTIFACT_PATH}" -T ${MIGRATION_FILE_NAME}

  artifacts:

    expire_in: 1 week

    paths:

      - ${MIGRATION_FILE_NAME}

  rules:

    - !reference [.all_environment_rules, rules]

  retry: !reference [.retry_rules]

 

# Apply the migration

.apply_cloud_migrate_template:

  image: ${BAR_REGISTRY}/mtb-docker-ga/${APPLY_MIGRATE_DB_IMAGE}

  stage: migrate_db

  tags:

    - 'cicd-k8s'

  variables:

    JOB_TOOL: db_migration

    JOB_FUNCTION: ef_migrations

    PROJECT_PLATFORM: "azure"

  # This may be able to be combined (maybe using some logic to check if it's a cloud db) but if I remember right,

  # the cli wouldn't remember the context for the az login so we daisy chained the command

  script:

    - !reference [.pipeline_scripts_linux, script]

    - cp ${NUGET_AUTH} ${NUGET_CONFIG_PATH}

    - |

      az login --service-principal --tenant "${ARM_TENANT_ID}" --username "${ARM_CLIENT_ID}" --password "${ARM_CLIENT_SECRET}"

      az account set -s ${ARM_SUBSCRIPTION_ID}

      dotnet ef database update --project ${PROJECT_FILE} ${VERBOSITY_LEVEL} --connection "${CONNECTION_STR}"

  retry: !reference [.retry_rules]

 

.dev_apply_cloud_migrate_db:

  extends: .apply_cloud_migrate_template

  environment: dev

  rules:

    - !reference [.lower_environment_rules, rules]

  when: manual
